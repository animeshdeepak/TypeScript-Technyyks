install node
check node version: node -v

install typescript globally: npm i typescript -g

transpile .ts file to .js file: tsc <folder-name>/<file-name.js>

you cant run .ts file like .js: node<folder-name>/<file-name.js> // error

run .ts code directly::

# 1. Remove the broken global versions
npm uninstall -g ts-node typescript
# 2. Create package.json if you don't have one
npm init -y
# 3. Install locally as dev dependencies
npm install -D typescript ts-node @types/node
npm install -D typescript@5 ts-node @types/node // version 5
# 4. Create tsconfig.json if you don't have one
npx tsc --init
# 5. Run your file
npx ts-node ts/index.ts
